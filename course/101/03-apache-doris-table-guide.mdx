    <CourseLessonUnit
        id="choose-table-model"
        label="Table models"
        title={<>What should repeated keys <em>mean</em>?</>}
        summary="Duplicate keeps every row, Unique keeps one logical row per key, and Aggregate combines value columns with declared functions."
        coachTitle="The model is a business-semantics choice."
        coach={<p>Use Duplicate for immutable facts and logs, Unique for CDC and latest-state records, and Aggregate when write-time rollups materially reduce repeated query work.</p>}
        calloutTitle="Hard to reverse"
        callout={<p>The table model cannot be changed after creation. Re-modeling requires a new table and data migration.</p>}

        label="Sort keys"
        title={<>Put common filters where storage can <em>use them</em>.</>}

    <CourseLessonUnit
        id="choose-bucketing"
        label="Bucketing"
        title={<>Distribute each partition into useful <em>tablets</em>.</>}
        summary="Hash bucketing gives deterministic placement by key; random bucketing spreads rows without requiring a distribution key."
        coachTitle="Balance locality, skew, and parallelism."
        coach={<p>Choose a high-cardinality, evenly distributed hash key when colocated access matters. Use random distribution when no reliable key exists and even write distribution is the priority.</p>}
        calloutTitle="Size before count"
        callout={<p>Estimate compressed data per partition, target healthy tablet sizes, and keep tablet count proportional to cluster resources. More buckets are not automatically faster.</p>}
        check={{
            question: 'What is the main risk of a low-cardinality hash bucket key?',
            options: ['Data skew', 'Automatic schema evolution'],
            correctIndex: 0,
            correctFeedback: 'Correct — too few distinct values can concentrate data in a subset of tablets.',
            incorrectFeedback: 'Bucketing affects distribution, not automatic schema evolution.',
